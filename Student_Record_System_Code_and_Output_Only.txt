071  int loadStudents(Student **students, int *capacity) {
072      FILE *fp = fopen(DATA_FILE, "rb");
073      Student item;
074      int count = 0;
075  
076      *capacity = INITIAL_CAPACITY;
077      *students = calloc((size_t)*capacity, sizeof(Student));
078      if (*students == NULL) {
079          printf("Memory allocation failed.\n");
080          return -1;
081      }
082  
083      if (fp == NULL) return 0; /* First run: data file does not exist yet. */
084  
085      while (fread(&item, sizeof(Student), 1, fp) == 1) {
086          if (count == *capacity) {
087              int newCapacity = (*capacity) * 2;
088              Student *temp = realloc(*students,
089                                      (size_t)newCapacity * sizeof(Student));
090              if (temp == NULL) {
091                  printf("Could not increase memory capacity.\n");
092                  fclose(fp);
093                  free(*students);
094                  *students = NULL;
095                  return -1;
096              }
097              *students = temp;
098              *capacity = newCapacity;
099          }
100          (*students)[count++] = item;
101      }
102  
103      if (ferror(fp)) printf("Warning: error while reading the data file.\n");
104      fclose(fp);
105      return count;
106  }

108  int saveStudents(const Student students[], int count) {
109      FILE *fp = fopen(DATA_FILE, "wb");
110      if (fp == NULL) {
111          printf("Unable to open %s for writing.\n", DATA_FILE);
112          return 0;
113      }
114      if (count > 0 && fwrite(students, sizeof(Student), (size_t)count, fp)
115                       != (size_t)count) {
116          printf("Error writing binary records.\n");
117          fclose(fp);
118          return 0;
119      }
120      if (fclose(fp) != 0) {
121          printf("Error closing binary file.\n");
122          return 0;
123      }
124      return 1;
125  }
